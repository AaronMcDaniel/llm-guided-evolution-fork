def calculate_accuracy(y_true, y_pred):
  """Calculates the accuracy of a classification model.

  Args:
    y_true: The true labels for the data.
    y_pred: The predicted labels for the data.

  Returns:
    The accuracy of the classification model.
  """
  return np.sum(y_true == y_pred) / len(y_true)
